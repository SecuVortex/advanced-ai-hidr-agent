### Example 2: Command-Line Testing

```python
from core.multiagent_monitor import MultiAgentHIDR

# Initialize
hidr = MultiAgentHIDR(require_human_approval=False)

# Analyze process
result = hidr.analyze_process(
    "suspicious.exe",
    "C:\\temp\\suspicious.exe",
    "",
    1234
)

# View results
hidr.print_summary(result)
hidr.print_agent_communication(result)
```

### Example 3: Automated Scanning

```python
import psutil
from core.multiagent_monitor import MultiAgentHIDR

hidr = MultiAgentHIDR()

# Scan all running processes
for proc in psutil.process_iter(['name', 'exe', 'cmdline', 'pid']):
    try:
        result = hidr.analyze_process(
            proc.info['name'],
            proc.info['exe'] or '',
            ' '.join(proc.info['cmdline'] or []),
            proc.info['pid']
        )
        
        if result['detection_result']['is_suspicious']:
            print(f"⚠️ Threat: {proc.info['name']}")
    except:
        pass
```
